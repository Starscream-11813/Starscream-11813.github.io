<script>
(function () {
  var asked = window.location.pathname;
  try { asked = decodeURI(asked); } catch (error) { /* keep it encoded */ }
  var suggestion = document.querySelector('[data-nf-suggestion]');
  if (!suggestion) { return; }

  /* Addresses compare without case, a trailing slash, ".html", or "index". */
  function norm(url) {
    return url.toLowerCase().replace(/index(\.html)?$/, '').replace(/\.html$/, '').replace(/\/+$/, '');
  }

  function distance(a, b) {
    var row = [], i, j, prev, next;
    for (j = 0; j <= b.length; j += 1) { row.push(j); }
    for (i = 1; i <= a.length; i += 1) {
      prev = row[0];
      row[0] = i;
      for (j = 1; j <= b.length; j += 1) {
        next = row[j];
        row[j] = Math.min(row[j] + 1, row[j - 1] + 1, prev + (a.charAt(i - 1) === b.charAt(j - 1) ? 0 : 1));
        prev = next;
      }
    }
    return row[b.length];
  }

  function say(html) {
    suggestion.innerHTML = html;
  }

  function link(page) {
    var a = document.createElement('a');
    a.href = page.url;
    a.textContent = page.title;
    return a.outerHTML;
  }

  if (norm(asked) === '/404') {
    return;
  }

  var pages = [];
  try {
    pages = JSON.parse(document.getElementById('nf-pages').textContent).filter(Boolean);
  } catch (error) {
    return;
  }

  var target = norm(asked);
  var last = target.split('/').pop();
  var section = target.split('/')[1] || '';
  var best = null;
  var ancestor = null;
  var containing = null;
  pages.forEach(function (page) {
    var url = norm(page.url);
    var d = distance(target, url);
    var same = url.split('/')[1] === section;
    if (!best || d < best.d) { best = { page: page, d: d }; }
    if (url && (target + '/').indexOf(url + '/') === 0 && (!ancestor || url.length > norm(ancestor.url).length)) { ancestor = page; }
    /* A page whose address holds the last part of yours, preferring your section (talks, books, ...) and then the shortest. */
    if (last.length >= 3 && url.split('/').pop().indexOf(last) !== -1 &&
        (!containing || (same && !containing.same) || (same === containing.same && url.length < norm(containing.page.url).length))) {
      containing = { page: page, same: same };
    }
  });

  if (best && best.d <= Math.max(2, Math.floor(target.length * 0.35))) {
    say('You were probably looking for ' + link(best.page) + ' (edit distance ' + best.d + ').');
  } else if (containing) {
    var quoted = document.createElement('code');
    quoted.textContent = last;
    say('You were probably looking for ' + link(containing.page) + ', whose address also contains ' + quoted.outerHTML + '.');
  } else if (ancestor) {
    say('You can return to ' + link(ancestor) + ', one level up.');
  } else {
    say('No close match found. Use the navigation above to find your way.');
  }
}());
</script>
